# plot.py — 由 nature-figure 生成
import matplotlib.pyplot as plt
import numpy as np, pandas as pd

df = pd.read_csv("assay_means.csv")
t  = ["0h", "24h", "48h"]
x  = np.arange(len(t)); w = 0.38

fig, ax = plt.subplots(figsize=(5.3, 3.6), dpi=300)
ax.bar(x - w/2, df.ctrl, w, yerr=df.ctrl_sd,
       label="对照", color="#2E7FB8", capsize=3)
ax.bar(x + w/2, df.drug, w, yerr=df.drug_sd,
       label="给药", color="#4FB49A", capsize=3)

ax.set_xticks(x, t); ax.legend(frameon=False)
fig.tight_layout()
fig.savefig("chart.png"); fig.savefig("chart.svg")
